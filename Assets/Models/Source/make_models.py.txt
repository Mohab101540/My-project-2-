# Builds low-poly Player, Thrower and Boss models and exports FBX into the Unity project.
# Run: blender -b -P make_models.py -- <models_dir> <preview_dir>
import bpy, math, sys, os

argv = sys.argv[sys.argv.index("--") + 1:]
OUT_DIR, PREVIEW_DIR = argv[0], argv[1]

bpy.ops.wm.read_factory_settings(use_empty=True)

MATS = {}
def mat(name, rgb, emit=0.0, metal=0.0, rough=0.6):
    if name in MATS:
        return MATS[name]
    m = bpy.data.materials.new(name)
    m.use_nodes = True
    b = m.node_tree.nodes.get("Principled BSDF")
    b.inputs["Base Color"].default_value = (*rgb, 1)
    b.inputs["Metallic"].default_value = metal
    b.inputs["Roughness"].default_value = rough
    if emit > 0:
        b.inputs["Emission Color"].default_value = (*rgb, 1)
        b.inputs["Emission Strength"].default_value = emit
    m.diffuse_color = (*rgb, 1)
    MATS[name] = m
    return m

def finish(obj, parent, material, bevel=0.0):
    obj.data.materials.append(material)
    obj.parent = parent
    if bevel > 0:
        mod = obj.modifiers.new("Bevel", "BEVEL")
        mod.width = bevel
        mod.segments = 1
        mod.limit_method = "ANGLE"
    for p in obj.data.polygons:
        p.use_smooth = False
    return obj

def box(name, parent, material, loc, size, rot=(0, 0, 0), bevel=0.02):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc, rotation=[math.radians(r) for r in rot])
    o = bpy.context.active_object
    o.name = name
    o.scale = size
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return finish(o, parent, material, bevel)

def cyl(name, parent, material, loc, r, depth, rot=(0, 0, 0), verts=10, bevel=0.0):
    bpy.ops.mesh.primitive_cylinder_add(vertices=verts, radius=r, depth=depth, location=loc,
                                        rotation=[math.radians(a) for a in rot])
    o = bpy.context.active_object
    o.name = name
    return finish(o, parent, material, bevel)

def sphere(name, parent, material, loc, r, scale=(1, 1, 1), seg=12, rings=8):
    bpy.ops.mesh.primitive_uv_sphere_add(segments=seg, ring_count=rings, radius=r, location=loc)
    o = bpy.context.active_object
    o.name = name
    o.scale = scale
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return finish(o, parent, material)

def cone(name, parent, material, loc, r1, r2, depth, rot=(0, 0, 0), verts=8):
    bpy.ops.mesh.primitive_cone_add(vertices=verts, radius1=r1, radius2=r2, depth=depth, location=loc,
                                    rotation=[math.radians(a) for a in rot])
    o = bpy.context.active_object
    o.name = name
    return finish(o, parent, material)

def root(name, x):
    bpy.ops.object.empty_add(location=(0, 0, 0))
    e = bpy.context.active_object
    e.name = name
    return e

# ---------------- EXPORT ----------------
def export(root_obj, filename):
    bpy.ops.object.select_all(action="DESELECT")
    root_obj.select_set(True)
    for c in root_obj.children_recursive:
        c.select_set(True)
    bpy.context.view_layer.objects.active = root_obj
    bpy.ops.export_scene.fbx(
        filepath=os.path.join(OUT_DIR, filename),
        use_selection=True,
        object_types={"EMPTY", "MESH"},
        apply_unit_scale=True,
        apply_scale_options="FBX_SCALE_UNITS",
        axis_forward="-Z",
        axis_up="Y",
        bake_space_transform=True,
        use_mesh_modifiers=True,
        mesh_smooth_type="FACE",
        add_leaf_bones=False,
    )

export(P, "Remodel_Player.fbx")
export(T, "Remodel_Thrower.fbx")
export(B, "Remodel_Boss.fbx")
